REPLACE:local_governor = {
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes
	
	employment_size = large_noble_employment
	
	category = government_category
	
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	allow = {
		custom_tooltip = {
			text = local_governor_tt
			scope:actor = {
				total_building_levels_including_construction:local_governor < {
					value = modifier:num_local_governors
					subtract = total_building_levels_including_construction:lieutenancy
				}
			}
		}
		owner = scope:actor #not in subjects
		trigger_if = { 
			limit = { has_building = building_type:seljuk_mint }
			NOT = { has_building = building_type:seljuk_mint }
		}
	}

	location_potential = {
		#has_road_to_capital = yes
		owner.capital != this
		NOT ={integration_level = conquered }
	}

	remove_if = {
		#location = { has_road_to_capital = no }
		location = { integration_level = conquered }
		#NOT = { owner = { at_war = yes } }
		#NOT = { owner = { in_civil_war = yes } }
	}

	expensive = yes
	
	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	important_for_AI = yes

	build_time = government_build_time


	unique_production_methods = {
		local_governor_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			fine_cloth = 1

			category = building_maintenance
		}
	}


	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 80
	}
}

REPLACE:order_commandery = {
	audio_tier = 3
	is_special = yes
	is_foreign = yes
	stronger_power_projection = no
	need_good_relation = yes
	pop_type = clergy

	category = religious_category

	max_levels = 1
	employment_size = order_religious_building_employment
	want_foreign_pop_created = no

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	
	allow_wrong_startup = yes
	
	
	location_potential = {
		custom_tooltip = {
			text = order_commandery_tt
			owner = {
				giving_scripted_relation = {
					type = relation_type:military_sponsorship
					target = scope:actor
				}
			}
		}
	}
	
	
	country_potential = {
		always = no
		modifier:target_of_military_sponsorships = yes
		religion ?= religion:catholic
	}
	
	build_time = soldier_building
	
	allow = {
		owner = { within_diplomatic_range = scope:actor }
		dominant_religion ?= religion:catholic
	}

	remove_if = {
		OR = {
			location = {
				dominant_religion != religion:catholic
			}
			owner = {
				OR = {
					modifier:target_of_military_sponsorships = no
					religion != religion:catholic
				}
			}	
		}
	}


	on_built = {
		if = {
			limit = {
				location.owner != root.owner 
			}
			location.owner = {
				add_opinion = { target = root.owner modifier = opinion_reformation_diplomacy } 
			}
		}
	}

	modifier = {
		gold_to_building_owner = 0.1
		manpower_to_building_owner = 0.005
	}
	construction_demand = order_commandery_construction
	possible_production_methods = {
		order_commandery_maintenance
	}
	graphical_tags = { small_military }
}
