# Event 1.1 - The Legacy of King Robert
provence.1 = {
	type = country_event
	fire_only_once = yes
	
	title = provence.1.title
	desc = provence.1.desc
	historical_info = provence.1.historical_info
	
	dynamic_historical_event = {
		tag = PRO
		from = 1337.1.1
		to = 1345.1.1
		monthly_chance = 50
	}
	
	illustration_tags = {
		10 = regular
		10 = interior
	}
	
	trigger = {
		tag = PRO
		NOT = { has_variable = pro_chain_1_started }
	}
	
	immediate = {
		set_variable = pro_chain_1_started
		event_illustration_estate_effect = { 
			foreground = estate_type:nobles_estate 
			background = estate_type:clergy_estate 
		}
		ruler = { save_scope_as = target_ruler }
		capital = { save_scope_as = target_location }
	}
	
	option = {
		name = provence.1.a
		tooltip = provence.1.a.tooltip
		historical_option = yes
		
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
		add_country_modifier = {
			modifier = pro_angevin_legitimacy
			years = 20
		}
		
		trigger_event_silently = { id = provence.2 days = 180 }
		
		ai_chance = {
			factor = 75
		}
	}
	
	option = {
		name = provence.1.b
		tooltip = provence.1.b.tooltip
		
		change_heir_selection = heir_selection:cognatic_primogeniture
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty

			modifier = pro_cognatic_succession_accepted
			years = 25
		}
		
		trigger_event_silently = { id = provence.2 days = 180 }
		
		ai_chance = {
			factor = 25
		}

}

	}

	}

			foreground = estate_type:nobles_estate 
			background = estate_type:nobles_estate 
		}
		
		create_named_dynasty = anjou_valois_dynasty
		
		create_character = {
			first_name = Rene
			dynasty = dynasty:anjou_valois_dynasty

			age = 20
			birth_location = location:angers
			female = no
			adm = 100
			dip = 90
			mil = 100
			add_trait = trait:scholar
			add_trait = trait:ambitious
			save_scope_as = new_ruler
		}
	}
	
	option = {
		name = provence.7.a
		tooltip = provence.7.a.tooltip
		historical_option = yes

		c:FRA ?= {
			add_opinion = { target = root modifier = opinion_supportive_monarch }
		}
